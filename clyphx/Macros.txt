# Spuren / tracks: "Synth 1", "Synth 2", "Synth 3", "Rhythm", "Mic", "LOOPER"

# --- Source select (User Action LLSRC) / Quelle wählen ----------------------
$LL_SYN1$ = LLSRC 1
$LL_SYN2$ = LLSRC 2
$LL_SYN3$ = LLSRC 3
$LL_RHY$  = LLSRC 4
$LL_MIC$  = LLSRC 5
$LL_NEXT$ = LLSRC >
$LL_PREV$ = LLSRC <
$LL_OFF$  = LLSRC OFF

# --- Same with built-in actions only (no User Action needed) ---------------
# --- Gleiches nur mit eingebauten Actions (ohne User Action) ---------------
$LLB_SYN1$ = "Synth 1", "Synth 2", "Synth 3"/ARM OFF ; "Synth 1"/ARM ON ; "LOOPER"/IN "Synth 1" ; "LOOPER"/INSUB "Post FX" ; "LOOPER"/MON IN
$LLB_SYN2$ = "Synth 1", "Synth 2", "Synth 3"/ARM OFF ; "Synth 2"/ARM ON ; "LOOPER"/IN "Synth 2" ; "LOOPER"/INSUB "Post FX" ; "LOOPER"/MON IN
$LLB_SYN3$ = "Synth 1", "Synth 2", "Synth 3"/ARM OFF ; "Synth 3"/ARM ON ; "LOOPER"/IN "Synth 3" ; "LOOPER"/INSUB "Post FX" ; "LOOPER"/MON IN
$LLB_RHY$  = "LOOPER"/IN "Rhythm" ; "LOOPER"/INSUB "Post FX" ; "LOOPER"/MON IN
$LLB_MIC$  = "Mic"/MON IN ; "LOOPER"/IN "Mic" ; "LOOPER"/INSUB "Post FX" ; "LOOPER"/MON IN

# --- Presets of the armed synth / Presets des aktiven Synths ---------------
# A) Program Change -> Pigments playlist (use the AU or VST2 version, not VST3).
#    ClyphX Pro Output = IAC "LL Synth", Track switch ON for "LL Synth" input.
#    PC 0 = first preset of the playlist / erstes Preset der Playlist.
$LL_PRE1$ = MIDI PC 1 0
$LL_PRE2$ = MIDI PC 1 1
$LL_PRE3$ = MIDI PC 1 2
$LL_PRE4$ = MIDI PC 1 3
$LL_PRE5$ = MIDI PC 1 4
$LL_PRE6$ = MIDI PC 1 5
$LL_PRE7$ = MIDI PC 1 6
$LL_PRE8$ = MIDI PC 1 7

# B) Instrument Rack, one chain per sound (rack = device 1 on the track).
#    Instrument Rack, eine Kette pro Sound (Rack = erstes Gerät der Spur).
$LL_RACK_NEXT$ = SEL/DEV(1) CSEL >
$LL_RACK_PREV$ = SEL/DEV(1) CSEL <
